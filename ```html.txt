<script>

    const articles = {

        science: {
            title: "Science",
            description:
                "Science is a systematic way of studying the natural world through observation, experimentation, measurement, and reasoning."
        },

        technology: {
            title: "Technology",
            description:
                "Technology refers to tools, techniques, systems, and processes created by people to solve problems and perform tasks."
        },

        history: {
            title: "History",
            description:
                "History is the study of past events, societies, cultures, and human experiences."
        },

        animals: {
            title: "Animals",
            description:
                "Animals are multicellular organisms that belong to the biological kingdom Animalia."
        }

    };


    function showArticle(id) {

        const articlesOnPage =
            document.querySelectorAll(".article");

        articlesOnPage.forEach(article => {
            article.classList.remove("active");
        });

        const selected =
            document.getElementById(id);

        if (selected) {
            selected.classList.add("active");
        }

        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });
    }


    function searchWiki() {

        const search =
            document.getElementById("searchInput")
            .value
            .toLowerCase()
            .trim();

        if (search === "") {
            showArticle("home");
            return;
        }

        const results =
            document.getElementById("results");

        results.innerHTML = "";

        let found = false;

        for (const key in articles) {

            const article = articles[key];

            if (
                article.title.toLowerCase().includes(search) ||
                article.description.toLowerCase().includes(search)
            ) {

                found = true;

                const result =
                    document.createElement("div");

                result.className = "result";

                result.innerHTML = `
                    <h3>${article.title}</h3>
                    <p>${article.description}</p>
                `;

                result.onclick = () =>
                    showArticle(key);

                results.appendChild(result);
            }
        }

        if (!found) {

            results.innerHTML = `
                <p>
                    No articles found for
                    <b>"${search}"</b>.
                </p>
            `;
        }

        showArticle("searchPage");
    }

</script>
